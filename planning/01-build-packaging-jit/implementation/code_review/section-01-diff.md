+	tool=${pair%%:*}

+		to_install+=("$formula")
+	fi

+	echo "+ brew install ${to_install[*]}"

+#!/bin/bash
+# Check that the tools Eikon's build needs are installed. macOS only.
+# Prints one line per tool and exits non-zero if anything is missing.
+# Never installs anything; see scripts/bootstrap.sh (make bootstrap).
+set -uo pipefail
+
+failed=0
+
+ok() { printf '  ok       %-12s %s\n' "$1" "$2"; }
+warn() { printf '  warning  %-12s %s\n' "$1" "$2"; }
+missing() {
+	printf '  MISSING  %-12s %s\n' "$1" "$2"
+	failed=1
+}
+
+have() { command -v "$1" >/dev/null 2>&1; }
+
+check_brew_tool() {
+	local tool=$1 formula=$2
+	if have "$tool"; then
+		ok "$tool" "$(command -v "$tool")"
+	else
+		missing "$tool" "brew install $formula (or make bootstrap)"
+	fi
+}
+
+echo "Eikon build tools:"
+
+# Xcode and the iOS SDK.
+if xcode-select -p >/dev/null 2>&1 && xcodebuild -version >/dev/null 2>&1; then
+	if sdk=$(xcrun --sdk iphoneos --show-sdk-path 2>/dev/null) && [ -n "$sdk" ]; then
+		ok xcode "$(xcodebuild -version | head -n 1), $(basename "$sdk")"
+	else
+		missing xcode "no iphoneos SDK found; install the iOS platform in Xcode settings"
+	fi
+else
+	missing xcode "install Xcode from the App Store, then run xcode-select -s /Applications/Xcode.app"
+fi
+
+check_brew_tool xcodegen xcodegen
+
+# ldid must be the Procursus build; Homebrew's "ldid" formula is an older one.
+if have ldid; then

+	fi

+#!/bin/bash
+# Version guard and build-version generator.
+#
+#   version.sh                  write build/generated/Version.xcconfig
+#   version.sh --check          every v* tag on HEAD must be v<VERSION>
+#   version.sh --check-tag TAG  TAG must be v<VERSION>
+#
+# Works on the git repository containing the current directory, not on the
+# script's own checkout. Runs on macOS /bin/bash 3.2 and on Linux bash.
+set -euo pipefail
+
+die() {
+	echo "version.sh: $*" >&2
+	exit 1
+}
+
+usage() {
+	die "usage: version.sh [--check | --check-tag <tag>]"
+}
+
+root=$(git rev-parse --show-toplevel 2>/dev/null) || die "not inside a git repository"
+
+read_version() {
+	[ -f "$root/VERSION" ] || die "VERSION file is missing"
+	local v
+	v=$(tr -d '[:space:]' <"$root/VERSION")
+	[[ "$v" =~ ^[0-9]+\.[0-9]+\.[0-9]+$ ]] || die "VERSION '$v' is not MAJOR.MINOR.PATCH"
+	echo "$v"
+}
+
+check_head_tags() {
+	local version tag
+	version=$(read_version)
+	git -C "$root" rev-parse --verify -q HEAD >/dev/null || die "repository has no commits"
+	for tag in $(git -C "$root" tag --points-at HEAD --list 'v*'); do
+		[ "$tag" = "v$version" ] || die "tag $tag on HEAD does not match VERSION $version"
+	done
+}
+
+check_tag() {
+	local version
+	version=$(read_version)
+	[ "$1" = "v$version" ] || die "tag $1 does not match VERSION $version (expected v$version)"
+}
+
+generate() {
+	local version build commit out_dir out tmp
+	version=$(read_version)
+	git -C "$root" rev-parse --verify -q HEAD >/dev/null || die "repository has no commits"
+
+	if [ -n "${EIKON_BUILD_NUMBER+set}" ]; then
+		[[ "$EIKON_BUILD_NUMBER" =~ ^[0-9]+$ ]] && [ "$EIKON_BUILD_NUMBER" -gt 0 ] ||
+			die "EIKON_BUILD_NUMBER '$EIKON_BUILD_NUMBER' is not a positive integer"
+		build=$EIKON_BUILD_NUMBER
+	else
+		if [ "$(git -C "$root" rev-parse --is-shallow-repository)" = "true" ]; then
+			die "shallow clone: the commit count would be wrong; use a full clone (fetch-depth: 0) or set EIKON_BUILD_NUMBER"
+		fi
+		build=$(git -C "$root" rev-list --count HEAD)
+	fi
+
+	commit=$(git -C "$root" rev-parse --short HEAD)
+	git -C "$root" update-index -q --refresh >/dev/null 2>&1 || true
+	if ! git -C "$root" diff-index --quiet HEAD --; then
+		commit="$commit-dirty"
+	fi
+
+	out_dir="$root/build/generated"
+	out="$out_dir/Version.xcconfig"
+	mkdir -p "$out_dir"
+	tmp=$(mktemp "$out_dir/.Version.xcconfig.XXXXXX")
+	trap 'rm -f "$tmp"' EXIT
+	cat >"$tmp" <<EOF
+// Generated by scripts/version.sh. Do not edit.
+MARKETING_VERSION = $version
+CURRENT_PROJECT_VERSION = $build
+EIKON_GIT_COMMIT = $commit
+EOF
+	mv -f "$tmp" "$out"
+	trap - EXIT
+	echo "Wrote $out ($version, build $build, $commit)"
+}
+
+case $# in
+0) generate ;;
+1)
+	case $1 in
+	--check) check_head_tags ;;
+	*) usage ;;
+	esac
+	;;
+2)
+	case $1 in
+	--check-tag) check_tag "$2" ;;
+	*) usage ;;
+	esac
+	;;
+*) usage ;;
+esac

+"""Shared pytest helpers for the Eikon scripts.
+
+Tests run the repo's scripts against throwaway git repositories, so they never
+depend on the state of the Eikon checkout itself.
+"""
+
+from __future__ import annotations
+
+import os
+import subprocess
+from pathlib import Path
+
+import pytest
+
+REPO_ROOT = Path(__file__).resolve().parent.parent
+
+# Identity and settings for fixture repos. Passed per command, so the user's
+# global git config is never read for identity or changed.
+_GIT_ENV = {
+    "GIT_AUTHOR_NAME": "Eikon Tests",
+    "GIT_AUTHOR_EMAIL": "tests@example.invalid",
+    "GIT_COMMITTER_NAME": "Eikon Tests",
+    "GIT_COMMITTER_EMAIL": "tests@example.invalid",
+    "GIT_CONFIG_NOSYSTEM": "1",
+}
+
+
+def git_env(extra: dict[str, str] | None = None) -> dict[str, str]:
+    env = {k: v for k, v in os.environ.items() if not k.startswith("GIT_")}
+    env.update(_GIT_ENV)
+    env.pop("EIKON_BUILD_NUMBER", None)
+    if extra:
+        env.update(extra)
+    return env
+
+
+def run_git(cwd: Path, *args: str) -> subprocess.CompletedProcess[str]:
+    return subprocess.run(
+        ["git", *args],
+        cwd=cwd,
+        env=git_env(),
+        check=True,
+        capture_output=True,
+        text=True,
+    )
+
+
+class GitRepo:
+    """A git repository in a temporary directory."""
+
+    def __init__(self, path: Path) -> None:
+        self.path = path
+
+    def git(self, *args: str) -> subprocess.CompletedProcess[str]:
+        return run_git(self.path, *args)
+
+    def write(self, relpath: str, text: str) -> Path:
+        target = self.path / relpath
+        target.parent.mkdir(parents=True, exist_ok=True)
+        target.write_text(text)
+        return target
+
+    def commit(self, message: str) -> None:
+        self.git("add", "-A")
+        self.git("commit", "--allow-empty", "-m", message)
+
+    def tag(self, name: str) -> None:
+        self.git("tag", name)
+
+
+def init_repo(path: Path) -> GitRepo:
+    path.mkdir(parents=True, exist_ok=True)
+    run_git(path, "init", "--initial-branch=main")
+    run_git(path, "config", "commit.gpgsign", "false")
+    run_git(path, "config", "tag.gpgsign", "false")
+    return GitRepo(path)
+
+
+@pytest.fixture
+def git_repo(tmp_path: Path) -> GitRepo:
+    """A new git repo with local identity configured. GitRepo offers
+    write(path, text), commit(message), tag(name), git(*args) and a .path attribute."""
+    return init_repo(tmp_path / "repo")
+
+
+def run_script(
+    script: str,
+    *args: str,
+    cwd: Path,
+    env: dict[str, str] | None = None,
+) -> subprocess.CompletedProcess[str]:
+    """Run a script from the Eikon checkout (for example "scripts/version.sh")
+    with its working directory set to `cwd`."""
+    return subprocess.run(
+        [str(REPO_ROOT / script), *args],
+        cwd=cwd,
+        env=git_env(env),
+        capture_output=True,
+        text=True,
+    )
